it("probe", () => {
  const out: string[] = [];
  const run = (id: string, cls: string, a: string, q: string) => {
    for (const general of [false, true]) {
      const v = validateAnswer({ answerText: a, question: q, general, served: true, refused: false, evidenceSufficient: true, sourceTexts: [] });
      out.push(JSON.stringify({ id, cls, lane: general ? "general" : "grounded", outcome: v.ok ? "ok" : `${v.kind}:${v.violation}` }));
    }
  };
  for (const [id, cls, a, q] of HAZ) run(id, cls, a, q);
  for (const [id, a, q] of BEN) run(id, "benign", a, q);
  writeFileSync("/private/tmp/claude-501/-Users-bravonode/6986cd8b-a127-455a-bf11-69c474826a74/scratchpad/probe-results.jsonl", out.join("\n") + "\n");
});
